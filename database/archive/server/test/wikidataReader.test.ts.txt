import { afterEach, beforeEach, expect, it, vi } from 'vitest';
import { getEventDateFromWikidata } from '../src/integrations/wikidataDate';

const fetchMock = vi.fn();
const response = (data: unknown, status = 200, headers: Record<string, string> = {}) => ({
  ok: status >= 200 && status < 300, status, headers: new Headers(headers), json: async () => data,
}) as Response;
beforeEach(() => { fetchMock.mockReset(); vi.stubGlobal('fetch', fetchMock); });
afterEach(() => { vi.useRealTimers(); vi.unstubAllGlobals(); });

it('normalizes IDs and distinguishes an absent date from a valid date', async () => {
  fetchMock.mockResolvedValueOnce(response({ results: { bindings: [] } }))
    .mockResolvedValueOnce(response({ results: { bindings: [{ date: { value: '+1944-06-06T00:00:00Z' } }] } }));
  expect(await getEventDateFromWikidata(' q123 ')).toBeNull();
  expect(new URL(fetchMock.mock.calls[0][0]).searchParams.get('query')).toContain('wd:Q123');
  expect(await getEventDateFromWikidata('Q123')).toBe('+1944-06-06T00:00:00Z');
});
it('rejects unsafe IDs before network access', async () => {
  await expect(getEventDateFromWikidata('Q123 } SERVICE')).rejects.toThrow('Q-id');
  expect(fetchMock).not.toHaveBeenCalled();
});
it.each([{}, { results: {} }, { results: { bindings: [{}] } }, { results: { bindings: [{ date: { value: 123 } }] } }, { error: 'failed', results: { bindings: [] } }])('rejects malformed responses instead of reporting no date: %j', async data => {
  fetchMock.mockResolvedValue(response(data));
  await expect(getEventDateFromWikidata('Q123')).rejects.toThrow('invalide');
  expect(fetchMock).toHaveBeenCalledOnce();
});
it('caps persistent rate-limit retries at three attempts', async () => {
  vi.useFakeTimers(); fetchMock.mockResolvedValue(response({}, 429));
  const failure = expect(getEventDateFromWikidata('Q123')).rejects.toThrow('indisponible');
  await vi.runAllTimersAsync(); await failure;
  expect(fetchMock).toHaveBeenCalledTimes(3);
});
it('respects Retry-After without retrying beyond the total deadline', async () => {
  fetchMock.mockResolvedValue(response({}, 503, { 'Retry-After': '60' }));
  await expect(getEventDateFromWikidata('Q123')).rejects.toThrow('indisponible');
  expect(fetchMock).toHaveBeenCalledOnce();
});
it('does not retry permanent HTTP failures', async () => {
  fetchMock.mockResolvedValue(response({}, 400));
  await expect(getEventDateFromWikidata('Q123')).rejects.toThrow('indisponible');
  expect(fetchMock).toHaveBeenCalledOnce();
});
it('bounds a hanging transport and aborts it after 30 seconds', async () => {
  vi.useFakeTimers(); fetchMock.mockReturnValue(new Promise(() => {}));
  const failure = expect(getEventDateFromWikidata('Q123')).rejects.toThrow('temps');
  await vi.advanceTimersByTimeAsync(30000); await failure;
  expect(fetchMock.mock.calls[0][1].signal.aborted).toBe(true);
});
it('allows caller cancellation to stop a hanging read immediately', async () => {
  const controller = new AbortController(); fetchMock.mockReturnValue(new Promise(() => {}));
  const failure = expect(getEventDateFromWikidata('Q123', { signal: controller.signal })).rejects.toMatchObject({ name: 'AbortError' });
  await vi.waitFor(() => expect(fetchMock).toHaveBeenCalledOnce());
  controller.abort(); await failure;
  expect(fetchMock.mock.calls[0][1].signal.aborted).toBe(true);
});
